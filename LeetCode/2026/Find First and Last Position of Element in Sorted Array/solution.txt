class Solution {

    public int first(int[] nums,int l,int h, int target){
        int ans=-1;
        while(l<=h){            
            int mid=(l+h)/2;

            if(nums[mid]==target){
                ans=mid;
                h=mid-1;
            }

            if(nums[mid]<target){
                l=mid+1;
            }

            if(nums[mid]>target){
                h=mid-1;
            }
        }
        return ans;
    }

    public int last(int[] nums, int l, int h, int target){
        int ans=-1;
        int f=first(nums,0,nums.length-1,target);
        if(f!=-1){
            while(l<=h){
                int mid=(l+h)/2;

                if(nums[mid]==target){
                    ans=mid;
                    l=mid+1;
                }
                if(nums[mid]<target){
                    l=mid+1;
                }
                if(nums[mid]>target){
                    h=mid-1;
                }
            }
            return ans;
        }
        return -1;

    }

    public int[] searchRange(int[] nums, int target) {
        List<Integer>result=new ArrayList<>();
        int f=first(nums,0,nums.length-1,target);
        int l=last(nums,0,nums.length-1,target);
        result.add(f);
        result.add(l);
        int[] result_arr = result.stream().mapToInt(Integer::intValue).toArray();

        return result_arr;


    }   
}
